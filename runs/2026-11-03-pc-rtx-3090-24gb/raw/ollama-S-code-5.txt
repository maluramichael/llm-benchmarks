def longest_common_prefix(strs):
    if not strs:
        return ""
    
    # Sort the strings so that if there is a common prefix,
    # it will be at the beginning of the first string and the end of the last.
    strs.sort()
    
    first = strs[0]
    last = strs[-1]
    
    i = 0
    while i < len(first) and i < len(last):
        if first[i] == last[i]:
            i += 1
        else:
            break
    
    return first[:i]
