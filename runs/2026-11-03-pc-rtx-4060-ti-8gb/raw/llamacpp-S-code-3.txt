def roman_to_int(s: str) -> int:
    """
    Converts a valid Roman numeral string (uppercase, value 1 to 3999) to an integer.
    
    Args:
        s (str): A valid Roman numeral string.
        
    Returns:
        int: The integer value of the Roman numeral.
    """
    roman_values = {
        'I': 1,
        'V': 5,
        'X': 10,
        'L': 50,
        'C': 100,
        'D': 500,
        'M': 1000
    }
    
    total = 0
    prev_value = 0
    
    # Iterate backwards to handle subtractive notation easily
    for char in reversed(s):
        current_value = roman_values[char]
        if current_value < prev_value:
            total -= current_value
        else:
            total += current_value
        prev_value = current_value
    
    return total
